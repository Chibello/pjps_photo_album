import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../../core/theme/app_colors.dart';
import '../../core/providers/providers.dart';
import '../../widgets/custom_text_field.dart';
import '../../models/year_level_model.dart';

class ManageYearsScreen extends ConsumerStatefulWidget {
  const ManageYearsScreen({super.key});

  @override
  ConsumerState<ManageYearsScreen> createState() => _ManageYearsScreenState();
}

class _ManageYearsScreenState extends ConsumerState<ManageYearsScreen> {
  List<YearLevelModel> _years = [];
  bool _isLoading = false;
  String? _selectedYearLevelId;

  @override
  void initState() {
    super.initState();
    _loadYearLevelsAndYears();
  }

  /// Load year levels and years dynamically from provider
  Future<void> _loadYearLevelsAndYears() async {
    setState(() => _isLoading = true);

    try {
      // Load year levels
      await ref.read(albumProvider.notifier).loadYearLevels();
      final yearLevels = ref.read(albumProvider).yearLevels;

      if (yearLevels.isNotEmpty) {
        _selectedYearLevelId ??= yearLevels.first.id;

        // Load years for selected year level
        await ref
            .read(albumProvider.notifier)
            .loadYearsForYearLevel(_selectedYearLevelId!);

        setState(() {
          _years = ref.read(albumProvider).currentYears;
        });
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Failed to load years: $e'),
          backgroundColor: AppColors.error,
        ),
      );
    } finally {
      setState(() => _isLoading = false);
    }
  }

  void _showAddYearDialog() {
    final nameController = TextEditingController();
    String? yearLevelId = _selectedYearLevelId;

    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Add New Year'),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            CustomTextField(controller: nameController, label: 'Year Name'),
            const SizedBox(height: 12),
            DropdownButtonFormField<String>(
              value: yearLevelId,
              decoration: const InputDecoration(
                labelText: 'Year Level',
                border: OutlineInputBorder(),
              ),
              items: ref.read(albumProvider).yearLevels.map((year) {
                return DropdownMenuItem(value: year.id, child: Text(year.name));
              }).toList(),
              onChanged: (value) => yearLevelId = value,
            ),
            const SizedBox(height: 12),
            DropdownButtonFormField<String>(
              value: '2023/2024',
              decoration: const InputDecoration(
                labelText: 'Academic Year',
                border: OutlineInputBorder(),
              ),
              items: const [
                DropdownMenuItem(value: '2023/2024', child: Text('2023/2024')),
                DropdownMenuItem(value: '2024/2025', child: Text('2024/2025')),
              ],
              onChanged: (value) {},
            ),
          ],
        ),
        actions: [
          TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('Cancel')),
          ElevatedButton(
            onPressed: () async {
              if (nameController.text.isEmpty || yearLevelId == null) return;

              Navigator.pop(context);

              // Call provider to add year (replace with backend logic if needed)
              await ref.read(albumProvider.notifier).addYear({
                'name': nameController.text,
                'yearLevelId': yearLevelId,
              });

              await _loadYearLevelsAndYears();

              ScaffoldMessenger.of(context).showSnackBar(
                const SnackBar(
                  content: Text('Year created successfully'),
                  backgroundColor: AppColors.success,
                ),
              );
            },
            child: const Text('Create'),
          ),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Manage Years'),
        actions: [
          IconButton(
            icon: const Icon(Icons.add),
            onPressed: _showAddYearDialog,
          ),
        ],
      ),
      body: _isLoading
          ? const Center(child: CircularProgressIndicator())
          : RefreshIndicator(
              onRefresh: _loadYearLevelsAndYears,
              child: ListView.builder(
                padding: const EdgeInsets.all(16),
                itemCount: _years.length,
                itemBuilder: (context, index) {
                  final yearItem = _years[index];
                  return Card(
                    margin: const EdgeInsets.only(bottom: 12),
                    child: ListTile(
                      leading: CircleAvatar(
                        backgroundColor: AppColors.primaryGold,
                        child: Text(yearItem.name[0],
                            style: const TextStyle(color: Colors.white)),
                      ),
                      title: Text(yearItem.name),
                      subtitle: Text(
                          '${yearItem.name} • ${yearItem.studentCount} students'),
                      trailing: Row(
                        mainAxisSize: MainAxisSize.min,
                        children: [
                          IconButton(
                            icon: const Icon(Icons.edit,
                                color: AppColors.primaryGold),
                            onPressed: () {
                              // TODO: edit year
                            },
                          ),
                          IconButton(
                            icon: const Icon(Icons.delete, color: Colors.red),
                            onPressed: () {
                              // TODO: delete year
                            },
                          ),
                        ],
                      ),
                    ),
                  );
                },
              ),
            ),
    );
  }
}
