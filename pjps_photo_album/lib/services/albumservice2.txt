import 'dart:io';
import 'package:dio/dio.dart';

import '../models/student_model.dart';
import '../models/class_model.dart';
import '../models/year_level_model.dart';
import 'api_service.dart';

class AlbumService {
  final ApiService _apiService;

  AlbumService(this._apiService);

  // =========================
  // Year Levels
  // =========================
  Future<List<YearLevelModel>> getYearLevels() async {
    try {
      final response = await _apiService.get('/albums/year-levels/');
      return (response.data as List)
          .map((json) => YearLevelModel.fromJson(json))
          .toList();
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // Classes
  // =========================
  Future<List<ClassGroupModel>> getClasses({String? yearLevelId}) async {
    try {
      final queryParams =
          yearLevelId != null ? {'year_level': yearLevelId} : null;

      final response = await _apiService.get(
        '/albums/classes/',
        queryParameters: queryParams,
      );

      return (response.data as List)
          .map((json) => ClassGroupModel.fromJson(json))
          .toList();
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  Future<ClassGroupModel> getClassDetails(String classId) async {
    try {
      final response = await _apiService.get('/albums/classes/$classId/');
      return ClassGroupModel.fromJson(response.data);
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // CREATE CLASS
  // =========================
  Future<ClassGroupModel> createClass(Map<String, dynamic> data) async {
    try {
      final response = await _apiService.post(
        '/albums/classes/',
        data: {
          'name': data['name'],
          'year_level': data['year_level'],
        },
      );

      return ClassGroupModel.fromJson(response.data);
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // Students
  // =========================
  Future<List<StudentListModel>> getStudentsInClass(String classId) async {
    try {
      final response =
          await _apiService.get('/albums/classes/$classId/students/');

      return (response.data as List)
          .map((json) => StudentListModel.fromJson(json))
          .toList();
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  Future<StudentDetailModel> getStudentDetails(String studentId) async {
    try {
      final response = await _apiService.get('/albums/students/$studentId/');
      return StudentDetailModel.fromJson(response.data);
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  Future<List<StudentListModel>> searchStudents(String query) async {
    try {
      final response = await _apiService.get(
        '/albums/students/',
        queryParameters: {'search': query},
      );

      return (response.data as List)
          .map((json) => StudentListModel.fromJson(json))
          .toList();
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // CREATE STUDENT
  // =========================
  Future<void> createStudent(Map<String, dynamic> data, File? image) async {
    try {
      final formData = FormData.fromMap({
        ...data,
        if (image != null)
          'profile_photo': await MultipartFile.fromFile(
            image.path,
            filename: image.path.split('/').last,
          ),
      });

      await _apiService.post(
        '/albums/students/',
        data: formData,
      );
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // UPDATE STUDENT
  // =========================
  Future<void> updateStudent(
      String id, Map<String, dynamic> data, File? image) async {
    try {
      final formData = FormData.fromMap({
        ...data,
        if (image != null)
          'profile_photo': await MultipartFile.fromFile(
            image.path,
            filename: image.path.split('/').last,
          ),
      });

      await _apiService.put(
        '/albums/students/$id/',
        data: formData,
      );
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // DELETE STUDENT
  // =========================
  Future<void> deleteStudent(String id) async {
    try {
      await _apiService.delete('/albums/students/$id/');
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // STAFF (🔥 FIXED & INSIDE CLASS)
  // =========================
  Future<List<dynamic>> getDepartments() async {
    try {
      final response = await _apiService.get('/staff/departments/');
      return response.data;
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  Future<List<dynamic>> getStaffCategories() async {
    try {
      final response = await _apiService.get('/staff/categories/');
      return response.data;
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  Future<void> createStaff(Map<String, dynamic> data, File? image) async {
    try {
      final formData = FormData.fromMap({
        ...data,
        if (image != null)
          'photo': await MultipartFile.fromFile(
            image.path,
            filename: image.path.split('/').last,
          ),
      });

      await _apiService.post('/staff/', data: formData);
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  Future<void> updateStaff(
      String id, Map<String, dynamic> data, File? image) async {
    try {
      final formData = FormData.fromMap({
        ...data,
        if (image != null)
          'photo': await MultipartFile.fromFile(
            image.path,
            filename: image.path.split('/').last,
          ),
      });

      await _apiService.put('/staff/$id/', data: formData);
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // Dashboard Stats
  // =========================
  Future<Map<String, dynamic>> getDashboardStats() async {
    try {
      final response = await _apiService.get('/albums/dashboard/stats/');
      return response.data;
    } on DioException catch (e) {
      throw Exception(_handleError(e));
    }
  }

  // =========================
  // Error Handler
  // =========================
  String _handleError(DioException e) {
    if (e.response?.data != null) {
      if (e.response?.data is Map) {
        return e.response?.data['error'] ??
            e.response?.data['message'] ??
            'An error occurred';
      }
      return e.response?.data.toString() ?? 'An error occurred';
    }
    return 'Network error. Please check your connection.';
  }
}
